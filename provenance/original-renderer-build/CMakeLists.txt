# Reuse the GPL-3.0 mesh renderer as a library. Do not install its global hooks,
# resource inspector, animation worker or separate framework plugin.

target_compile_options(food_renderer PRIVATE /utf-8 /Zc:preprocessor /permissive- /W3)

target_include_directories(LocalFoodGallery PRIVATE src)
target_precompile_headers(LocalFoodGallery PRIVATE src/PCH.h)
target_link_libraries(LocalFoodGallery PRIVATE food_renderer)
target_compile_options(LocalFoodGallery PRIVATE /utf-8 /Zc:preprocessor /permissive- /W4 /Zi)
target_link_options(LocalFoodGallery PRIVATE /DEBUG /OPT:REF /OPT:ICF)
